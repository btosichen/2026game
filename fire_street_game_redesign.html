📊 Google 試算表 (GAS) 排行榜設定

將遊戲分數與生存時間自動記錄到 Google 試算表！

// Google Apps Script 程式碼範例：
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < data.length; i++) {
    list.push({ name: data[i][0], score: data[i][1], time: data[i][2], date: data[i][3] });
  }
  list.sort(function(a, b) { return b.score - a.score; });
  return ContentService.createTextOutput(JSON.stringify(list.slice(0, 10)))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var p = JSON.parse(e.postData.contents);
  sheet.appendRow([p.name, p.score, p.time, new Date().toLocaleString()]);
  return ContentService.createTextOutput(JSON.stringify({status: "ok"}))
    .setMimeType(ContentService.MimeType.JSON);
}
